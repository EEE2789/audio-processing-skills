# 错误反省记录 - generate-cover-v2

## 2026-03-11：`--title` 参数格式错误

### 错误场景
用户要求生成标题为"反抽加空单"的封面。

### 错误操作
```bash
python generate_cover.py --auto --title "反抽加空单"
```

### 问题原因
- 脚本只识别 `--title=xxx` 格式（等号连接）
- `--title "xxx"` 格式会被解析为两个独立参数：
  - `--title` - 不匹配任何已知参数，被忽略
  - `"反抽加空单"` - 不是 .txt 文件、不是颜色、不是日期，被跳过
- 结果：脚本使用了 AI 生成的标题而非用户指定的标题

### 正确操作
```bash
python generate_cover.py --auto --title="反抽加空单"
```

### 教训
1. **传递用户指定标题时，必须使用 `--title=标题` 格式（等号，无空格）**
2. 执行后检查输出中的 `📝 使用指定标题:` 确认标题是否被正确使用
3. 不要假设参数格式，第一次执行时验证输出是否符合预期
4. 如果输出显示 `🤖 自动模式：已选择第1个标题:` 而非 `📝 使用指定标题:`，说明参数格式错误

### 代码位置
脚本参数解析逻辑在 `generate_cover.py` 第 498-504 行：
```python
for i, arg in enumerate(sys.argv[1:]):
    if arg.startswith('--title='):
        custom_title = arg.split('=', 1)[1]

```

- 前置 0.2 秒封面

- 字幕已烧录在 Video4 中，无需额外处理

- 检查 Video4 是否存在
- 如果不存在，自动生成 Video4
- Video4 作为以太坊视频的源文件

#!/usr/bin/env python3
"""
安装定时清理任务

使用 macOS launchd 配置每天早上 6 点（北京时间）自动清理视频管道目录。

注意：macOS 使用本地时区，如果需要北京时间早上 6 点，需要根据系统时区调整。
"""

import os
import subprocess
import plistlib
from pathlib import Path


# ====== 配置 ======
SCRIPT_PATH = "/Users/ai/.claude/skills/clean-pipeline/scripts/clean.py"
PLIST_NAME = "com.videopipeline.clean.plist"
LAUNCH_AGENTS_DIR = Path.home() / "Library" / "LaunchAgents"
PLIST_PATH = LAUNCH_AGENTS_DIR / PLIST_NAME


# 北京时间早上 6 点对应的 macOS 本地时间（假设系统时区为 UTC+8）
# 如果系统时区不同，需要调整小时数
HOUR = 6
MINUTE = 0


def create_plist():
    """创建 launchd plist 文件"""
    plist = {
        "Label": "com.videopipeline.clean",
        "ProgramArguments": [
            "/usr/bin/python3",
            SCRIPT_PATH
        ],
        "StartCalendarInterval": {
            "Hour": HOUR,
            "Minute": MINUTE
        },
        "StandardOutPath": str(Path.home() / "Library" / "Logs" / "videopipeline_clean.log"),
        "StandardErrorPath": str(Path.home() / "Library" / "Logs" / "videopipeline_clean_err.log"),
        "RunAtLoad": False,  # 不在加载时立即运行
    }

    # 确保 LaunchAgents 目录存在
    LAUNCH_AGENTS_DIR.mkdir(parents=True, exist_ok=True)

    # 写入 plist 文件
    with open(PLIST_PATH, "wb") as f:
        plistlib.dump(plist, f)

    print(f"✅ plist 文件已创建: {PLIST_PATH}")
    return PLIST_PATH


def load_plist():
    """加载 launchd 任务"""
    try:
        result = subprocess.run(
            ["launchctl", "load", str(PLIST_PATH)],
            capture_output=True,
            text=True
        )
        if result.returncode == 0:
            print(f"✅ 定时任务已加载")
            print(f"⏰ 将在每天 {HOUR:02d}:{MINUTE:02d} 自动清理视频管道目录")
            return True
        else:
            print(f"❌ 加载失败: {result.stderr}")
            return False
    except Exception as e:
        print(f"❌ 加载失败: {e}")
        return False


def unload_plist():
    """卸载 launchd 任务"""
    try:
        result = subprocess.run(
            ["launchctl", "unload", str(PLIST_PATH)],
            capture_output=True,
            text=True
        )
        if result.returncode == 0:
            print(f"✅ 定时任务已卸载")
            return True
        else:
            print(f"❌ 卸载失败: {result.stderr}")
            return False
    except Exception as e:
        print(f"❌ 卸载失败: {e}")
        return False


def list_jobs():
    """列出所有 videopipeline 相关任务"""
    try:
        result = subprocess.run(
            ["launchctl", "list"],
            capture_output=True,
            text=True
        )
        if result.returncode == 0:
            lines = result.stdout.split('\n')
            for line in lines:
                if 'videopipeline' in line.lower() or 'com.videopipeline' in line:
                    print(f"  {line}")
    except Exception as e:
        print(f"❌ 列出任务失败: {e}")


def main():
    import argparse

    parser = argparse.ArgumentParser(description="安装/卸载定时清理任务")
    parser.add_argument(
        "--install",
        action="store_true",
        help="安装定时任务（每天早上 6 点执行）"
    )
    parser.add_argument(
        "--uninstall",
        action="store_true",
        help="卸载定时任务"
    )
    parser.add_argument(
        "--list",
        action="store_true",
        help="列出相关定时任务"
    )
    parser.add_argument(
        "--hour",
        type=int,
        default=HOUR,
        help="执行小时（默认：6）"
    )
    parser.add_argument(
        "--minute",
        type=int,
        default=MINUTE,
        help="执行分钟（默认：0）"
    )

    args = parser.parse_args()

    # 更新时间配置
    global HOUR, MINUTE
    HOUR = args.hour
    MINUTE = args.minute

    print("=" * 60)
    print("视频管道定时清理任务配置")
    print("=" * 60)

    if args.list:
        print("\n📋 相关定时任务:")
        list_jobs()
    elif args.uninstall:
        print(f"\n🗑️  卸载定时任务...")
        unload_plist()
        # 删除 plist 文件
        if PLIST_PATH.exists():
            PLIST_PATH.unlink()
            print(f"✅ 已删除 plist 文件: {PLIST_PATH}")
    elif args.install:
        print(f"\n📦 安装定时任务...")
        print(f"⏰ 执行时间: 每天 {HOUR:02d}:{MINUTE:02d}")
        print(f"📂 清理脚本: {SCRIPT_PATH}")
        print()
        create_plist()
        load_plist()
        print("\n💡 提示:")

def load_env():
    """加载环境变量"""

    return {

async def listen_and_download(env_config):
    """监听 Telegram 并下载视频"""
    try:
        from telethon import TelegramClient
        from telethon.events import NewMessage
    except ImportError:
        print("❌ Telethon 未安装")
        if not install_telethon():
            return
        from telethon import TelegramClient
        from telethon.events import NewMessage

    DOWNLOAD_DIR.mkdir(parents=True, exist_ok=True)

    # 加载已下载视频列表
    downloaded_videos = load_downloaded_videos()
    print(f"📋 已记录 {len(downloaded_videos)} 个已下载视频")

    print(f"\n📱 正在连接 Telegram...")
    print(f"   手机号: {env_config['phone']}")
    print(f"   监听频道: {', '.join(env_config['channels'])}")
    print(f"   保存到: {DOWNLOAD_DIR}")

    client = TelegramClient(
        str(SESSION_FILE),
        env_config['api_id'],
        env_config['api_hash']
    )

    @client.on(NewMessage(chats=env_config['channels'], func=lambda e: e.message.video))
    async def handle_video(event):
        """处理新视频消息"""
        message = event.message
        video_id = str(message.id)

        # 检查是否已下载
        if video_id in downloaded_videos:
            print(f"⏭️  视频 {video_id} 已下载，跳过")
            return

        try:
            # 生成文件名
            now = datetime.now()
            date_str = now.strftime("%m%d")

            # 获取原始文件名
            from telethon.tl.types import DocumentAttributeFilename
            original_name = None
            if message.video.attributes:
                for attr in message.video.attributes:
                    if isinstance(attr, DocumentAttributeFilename):
                        original_name = attr.file_name
                        break

            if original_name and original_name.endswith('.mp4'):
                filename = original_name
            else:
                filename = f"{date_str}.mp4"

            file_path = DOWNLOAD_DIR / filename

            print(f"\n📥 检测到新视频: {filename}")
            print(f"   来源: {event.chat.title}")
            print(f"   大小: {message.video.size / (1024*1024):.1f} MB")
            print(f"   正在下载...")

            # 下载视频
            await client.download_media(message.video, file_path)

            # 记录已下载
            save_downloaded_video(video_id)
            downloaded_videos.add(video_id)

            print(f"   ✅ 下载完成: {filename}")

            # 发送系统通知
            time_str = now.strftime("%H:%M")
            send_notification(
                "Telegram 视频下载",
                f"{filename} 已下载 ({time_str})"
            )

        except Exception as e:
            print(f"   ❌ 下载失败: {e}")

    # 启动客户端
    await client.start(phone=env_config['phone'])

    print("✅ 已连接到 Telegram")
    print("🎧 正在监听视频消息...")
    print("💡 按 Ctrl+C 停止监听\n")

    # 保持运行
    try:
        await client.run_until_disconnected()
    except KeyboardInterrupt:
        print("\n\n⏹️  监听已停止")

#!/usr/bin/env python3
"""
安装 Telegram 监听 Bot 为常驻服务

使用 macOS launchd 配置开机自启动和常驻运行。
"""

import os
import subprocess
import plistlib
from pathlib import Path


# ====== 配置 ======
SCRIPT_PATH = "/Users/ai/.claude/skills/telegram-download/scripts/listen_and_download.py"
PLIST_NAME = "com.telegram.videolistener.plist"
LAUNCH_AGENTS_DIR = Path.home() / "Library" / "LaunchAgents"
PLIST_PATH = LAUNCH_AGENTS_DIR / PLIST_NAME


def create_plist():
    """创建 launchd plist 文件"""
    plist = {
        "Label": "com.telegram.videolistener",
        "ProgramArguments": [
            "/usr/bin/python3",
            SCRIPT_PATH
        ],
        "RunAtLoad": True,  # 加载时立即运行
        "KeepAlive": {
            "SuccessfulExit": False,  # 正常退出不重启
            "Crashed": True,  # 崩溃时重启
        },
        "StandardOutPath": str(Path.home() / "Library" / "Logs" / "telegram_video_listener.log"),
        "StandardErrorPath": str(Path.home() / "Library" / "Logs" / "telegram_video_listener_err.log"),
        "WorkingDirectory": str(Path.home()),
    }

    # 确保 LaunchAgents 目录存在
    LAUNCH_AGENTS_DIR.mkdir(parents=True, exist_ok=True)

    # 写入 plist 文件
    with open(PLIST_PATH, "wb") as f:
        plistlib.dump(plist, f)

    print(f"✅ plist 文件已创建: {PLIST_PATH}")
    return PLIST_PATH


def load_plist():
    """加载 launchd 任务"""
    try:
        result = subprocess.run(
            ["launchctl", "load", str(PLIST_PATH)],
            capture_output=True,
            text=True
        )
        if result.returncode == 0:
            print(f"✅ 监听任务已加载并启动")
            return True
        else:
            print(f"❌ 加载失败: {result.stderr}")
            return False
    except Exception as e:
        print(f"❌ 加载失败: {e}")
        return False


def unload_plist():
    """卸载 launchd 任务"""
    try:
        # 先尝试停止任务
        subprocess.run(
            ["launchctl", "stop", "com.telegram.videolistener"],
            capture_output=True
        )
    except:
        pass

    try:
        result = subprocess.run(
            ["launchctl", "unload", str(PLIST_PATH)],
            capture_output=True,
            text=True
        )
        if result.returncode == 0:
            print(f"✅ 监听任务已停止并卸载")
            return True
        else:
            print(f"❌ 卸载失败: {result.stderr}")
            return False
    except Exception as e:
        print(f"❌ 卸载失败: {e}")
        return False


def list_jobs():
    """列出所有 telegram 相关任务"""
    try:
        result = subprocess.run(
            ["launchctl", "list"],
            capture_output=True,
            text=True
        )
        if result.returncode == 0:
            lines = result.stdout.split('\n')
            for line in lines:
                if 'telegram' in line.lower() or 'videolistener' in line.lower():
                    print(f"  {line}")
    except Exception as e:
        print(f"❌ 列出任务失败: {e}")

#!/usr/bin/env python3
"""
Telegram 视频监听 Bot - 服务器版本

监听指定 Telegram 群组/频道，当检测到新视频时自动下载。
运行在服务器上，然后通过 rsync 同步到本地。
"""

import os
import sys
import subprocess
import asyncio
from pathlib import Path
from datetime import datetime

# ====== 配置 ======
SCRIPT_DIR = Path("/root/telegram_video_bot")
ENV_FILE = SCRIPT_DIR / ".env"
DOWNLOAD_DIR = Path("/root/videos")
DOWNLOADED_VIDEOS_FILE = SCRIPT_DIR / ".downloaded_videos.txt"
LOG_FILE = SCRIPT_DIR / "listener.log"

# 默认配置
DEFAULT_CHANNELS = ["-1001867687861"]  # 禁言群


def load_env():
    """加载环境变量"""
    if not ENV_FILE.exists():
        print(f"❌ 环境变量文件不存在: {ENV_FILE}")
        return {}

    import json
    try:
        with open(ENV_FILE, 'r') as f:
            return json.load(f)
    except:
        return {}


def check_telethon_installed():
    """检查 telethon 是否已安装"""
    try:
        import telethon
        return True
    except ImportError:
        return False


def install_telethon():
    """安装 telethon"""
    print("📦 正在安装 Telethon...")
    try:
        result = subprocess.run(
            [sys.executable, "-m", "pip", "install", "telethon"],
            capture_output=True,
            text=True
        )
        if result.returncode == 0:
            print("✅ Telethon 安装成功")
            return True
        else:
            print(f"❌ Telethon 安装失败: {result.stderr}")
            return False
    except Exception as e:
        print(f"❌ Telethon 安装失败: {e}")
        return False


def load_downloaded_videos():
    """加载已下载的视频 ID 列表"""
    if DOWNLOADED_VIDEOS_FILE.exists():
        with open(DOWNLOADED_VIDEOS_FILE, 'r') as f:
            return set(line.strip() for line in f if line.strip())
    return set()


def save_downloaded_video(video_id):
    """保存已下载的视频 ID"""
    with open(DOWNLOADED_VIDEOS_FILE, 'a') as f:
        f.write(f"{video_id}\n")


def log(message):
    """记录日志"""
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    with open(LOG_FILE, 'a') as f:
        f.write(f"[{timestamp}] {message}\n")
    print(message)


async def listen_and_download(channels):
    """监听 Telegram 并下载视频"""
    try:
        from telethon import TelegramClient
        from telethon.events import NewMessage
    except ImportError:
        print("❌ Telethon 未安装")
        if not install_telethon():
            return
        from telethon import TelegramClient
        from telethon.events import NewMessage

    # 确保目录存在
    DOWNLOAD_DIR.mkdir(parents=True, exist_ok=True)
    SCRIPT_DIR.mkdir(parents=True, exist_ok=True)

    # 创建会话文件
    session_name = str(SCRIPT_DIR / ".telegram_listener_session")

    # 加载环境变量
    env_config = load_env()
    api_id = env_config.get('API_ID')
    api_hash = env_config.get('API_HASH')
    phone = env_config.get('PHONE')

    if not api_id or not api_hash or not phone:
        print("❌ 请在 .env 文件中配置 API_ID、API_HASH 和 PHONE_NUMBER")
        return

    log(f"正在连接 Telegram...")
    log(f"手机号: {phone}")
    log(f"监听频道: {channels}")

    client = TelegramClient(
        session_name,
        int(api_id),
        api_hash
    )

    # 加载已下载视频列表
    downloaded_videos = load_downloaded_videos()
    log(f"已记录 {len(downloaded_videos)} 个已下载视频")

    @client.on(NewMessage(chats=channels, func=lambda e: e.message.video))
    async def handle_video(event):
        """处理新视频消息"""
        message = event.message
        video_id = str(message.id)

        # 检查是否已下载
        if video_id in downloaded_videos:
            log(f"⏭️  视频 {video_id} 已下载，跳过")
            return

        try:
            # 生成文件名
            now = datetime.now()
            date_str = now.strftime("%m%d")

            # 获取原始文件名
            from telethon.tl.types import DocumentAttributeFilename
            original_name = None
            if message.video.attributes:
                for attr in message.video.attributes:
                    if isinstance(attr, DocumentAttributeFilename):
                        original_name = attr.file_name
                        break

            if original_name and original_name.endswith('.mp4'):
                filename = original_name
            else:
                filename = f"{date_str}.mp4"

            file_path = DOWNLOAD_DIR / filename

            log(f"📥 检测到新视频: {filename}")
            log(f"   来源: {event.chat.title}")
            log(f"   大小: {message.video.size / (1024*1024):.1f} MB")
            log(f"   正在下载...")

            # 下载视频
            await client.download_media(message.video, file_path)

            # 记录已下载
            save_downloaded_video(video_id)
            downloaded_videos.add(video_id)

            log(f"✅ 下载完成: {filename}")
            log(f"   保存路径: {file_path}")

        except Exception as e:
            log(f"❌ 下载失败: {e}")

    # 启动客户端
    await client.start(phone=phone)

    log("✅ 已连接到 Telegram")
    log("🎧 正在监听视频消息...")
    log("💡 按 Ctrl+C 停止监听\n")

    # 保持运行
    try:
        await client.run_until_disconnected()
    except KeyboardInterrupt:
        log("\n⏹️  监听已停止")
    except Exception as e:
        log(f"❌ 错误: {e}")


def main():
    """主函数"""
    import argparse

    parser = argparse.ArgumentParser(description="Telegram 视频监听 Bot - 服务器版本")
    parser.add_argument(
        "--install",
        action="store_true",
        help="安装依赖"
    )

    args = parser.parse_args()

    print("=" * 60)
    print("Telegram 视频监听 Bot - 服务器版本")
    print("=" * 60)

    if args.install:
        print("\n📦 安装依赖...")
        install_telethon()
        return

    # 加载配置
    env_config = load_env()
    channels = env_config.get('CHANNELS', DEFAULT_CHANNELS)

    if isinstance(channels, str):
        channels = [c.strip() for c in channels.split(',') if c.strip()]

    print(f"\n📋 监听频道: {channels}")
    print(f"📂 下载目录: {DOWNLOAD_DIR}")
    print(f"📋 日志文件: {LOG_FILE}")

    # 运行监听
    try:
        asyncio.run(listen_and_download(channels))
    except KeyboardInterrupt:
        print("\n👋 再见！")
    except Exception as e:
        print(f"\n❌ 错误: {e}")
        import traceback
        traceback.print_exc()
        sys.exit(1)


if __name__ == "__main__":
    main()
